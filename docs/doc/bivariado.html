Ahora, realizaremos el análisis de las ganancias según estado

df %>% 
  ggplot(aes(x = State,y=Profit))+
  geom_boxplot()
# === 1. Calcular atípicos de forma más robusta ===
atipicos <- df %>%
  group_by(State) %>%
  mutate(
    Q1 = quantile(Profit, 0.25, na.rm = TRUE),
    Q3 = quantile(Profit, 0.75, na.rm = TRUE),
    IQR = Q3 - Q1,
    lower_limit = Q1 - 1.5 * IQR,
    upper_limit = Q3 + 1.5 * IQR
  ) %>%
  filter(Profit < lower_limit | Profit > upper_limit) %>%
  ungroup()


p <- df %>%
  ggplot(aes(x = State, y = Profit, fill = State)) +
  
  geom_boxplot(alpha = 0.75, 
               outlier.shape = NA, 
               width = 0.6, 
               linewidth = 0.8) +
  
  # Solo puntos atípicos en rojo
  geom_point(data = atipicos,
             aes(x = State, y = Profit),
             color = "#E74C3C", 
             size = 3.2, 
             alpha = 0.9) +
  
  stat_summary(fun = mean, 
               geom = "point", 
               shape = 16, 
               size = 4,

  scale_fill_manual(values = c("New York" = "#AED6F1", 
                               "California" = "#A9DFBF",

  scale_y_continuous(labels = scales::dollar_format()) +
  
  labs(title = "Distribución del Profit por Estado",

       y = "Profit (USD)",

  theme_minimal(base_size = 13) +
