1.1 Análisis exploratorio de Profit

Veamos inicialmente nuestra variable respuesta, que es el profit

# Desactivar la notación científica para todo lo que sigue
options(scipen = 999)

df %>% 
  summarise(n = length(Profit),
            prom = mean(Profit),
            ds = sd(Profit),
            mediana = median(Profit),
            RIC = IQR(Profit),
            min = min(Profit),
            max = max(Profit)) -> uniprofit

uniprofit%>% 
  kable()
n prom ds mediana RIC min max
50 112012.6 40306.18 107978.2 49627.07 14681.4 192261.8

De las 50 startups de EEUU, el promedio de ganancia fue 112012.6 usd (ds = 40306.2 usd), donde el 50% de las startups tienen ganancias menor o igual a 107978.2usd (RIC = 49627.1 usd). Además, el mínino de ganancias de las startups es de 14681.4 usd y un maximo de 192261.8 usd.

df %>% 
  ggplot(aes(x = "", y = Profit)) + # Usamos "" en x para centrar el boxplot único
  
  # Boxplot estético con color pastel (#A8DADC es un azul pastel muy limpio)
  geom_boxplot(
    fill = "#A8DADC", 
    color = "#1D3557", 
    alpha = 0.8, 
    width = 0.5
  ) +
  
  # Añadir el PROMEDIO de forma dinámica (como un rombo rojo)
  stat_summary(
    fun = mean, 
    geom = "point", 
    shape = 18, 
    size = 4, 
    color = "#E63946"
  ) +
  
  # Limpieza de etiquetas
  labs(
    title = "Distribución del Beneficio (Profit)",
    x = NULL, # Removemos la etiqueta del eje X que no aporta nada
    y = "Profit ($)"
  ) +
  
  # Tema profesional y limpio
  theme_minimal(base_size = 14) +
  theme(
    plot.title = element_text(face = "bold", hjust = 0.5),
    plot.subtitle = element_text(hjust = 0.5, color = "gray40"),
    panel.grid.major.x = element_blank() # Quita la línea de cuadrícula del eje X
  )

1.2 Análisis exploratorio de las caracteristica

Realicemos un EDA de las variables numéricas

df %>% 
  summarise(n = length(R.D.Spend),
            prom = mean(R.D.Spend),
            ds = sd(R.D.Spend),
            mediana = median(R.D.Spend),
            RIC = IQR(R.D.Spend),
            min = min(R.D.Spend),
            max = max(R.D.Spend)) -> uninvest

df %>% 
  summarise(n = length(Administration),
            prom = mean(Administration),
            ds = sd(Administration),
            mediana = median(Administration),
            RIC = IQR(Administration),
            min = min(Administration),
            max = max(Administration)) -> uniadm

df %>% 
  summarise(n = length(Marketing.Spend),
            prom = mean(Marketing.Spend),
            ds = sd(Marketing.Spend),
            mediana = median(Marketing.Spend),
            RIC = IQR(Marketing.Spend),
            min = min(Marketing.Spend),
            max = max(Marketing.Spend)) -> unimark

uninvest$Variable <- "Investigación"
uniadm$Variable <- "Administración"
unimark$Variable <- "Marketing"

bind_rows(uninvest, uniadm, unimark) %>%
  select(Variable, everything()) %>%

  row_spec(0, bold = TRUE, background = "#2C3E50", color = "white") %>%
